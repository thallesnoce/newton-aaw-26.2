O BUG PLANTADO
--------------
Arquivo: src/ResilientOrders.Api/Services/PedidoService.cs
Método:  CalcularDesconto

    if (subtotal > 500m)      // <-- BUG: deveria ser >= 500m

A regra do handout diz "10% a partir de R$500 (inclusive)".
Com o bug, subtotal EXATAMENTE 500 cai na faixa de 5%:
    esperado: 500 - 50 (10%) + 0 (frete grátis) = 450,00
    obtido:   500 - 25 (5%)  + 0               = 475,00
Só o caso de borda subtotal = 500 revela o problema — é o coração da aula.

CORREÇÃO
--------
    if (subtotal >= 500m)
    {
        return Math.Round(subtotal * 0.10m, 2);
    }

[Fact(DisplayName = "CriarPedido: quantidade <= 0 lança ArgumentException")]
public void CriarPedido_ComQuantidadeZero_LancaArgumentException()
{

    var mockRepo = new Mock<IPedidoRepositorio>();
    var service = new PedidoService(mockRepo.Object);
    var request = new CriarPedidoRequest("Maria", new List<ItemPedido>
    {
        new() { ProdutoId = 1, NomeProduto = "Mouse", PrecoUnitario = 50m, Quantidade = 0 }

public void CriarPedido_Valido_SalvaUmaVez()
{

    var service = new PedidoService(mockRepo.Object);
    var request = new CriarPedidoRequest("Maria", new List<ItemPedido>
    {
        new() { ProdutoId = 1, NomeProduto = "Teclado", PrecoUnitario = 100m, Quantidade = 2 }
    });

    // Act
    var pedido = service.CriarPedido(request);

    // Assert
    pedido.Subtotal.Should().Be(200m);
    pedido.Desconto.Should().Be(10m);
    pedido.Frete.Should().Be(25m);
    pedido.Total.Should().Be(215m);
    mockRepo.Verify(r => r.Salvar(It.IsAny<Pedido>()), Times.Once);
}

[Fact(DisplayName = "POST /api/pedidos: 201 válido e 400 sem itens")]
public async Task PostPedidos_Valido201_Invalido400()
{
    // Arrange
    var valido = new
    {
        nomeCliente = "Maria",
        itens = new[] { new { produtoId = 1, nomeProduto = "Mouse", precoUnitario = 50m, quantidade = 2 } }
    };
    var invalido = new { nomeCliente = "Maria", itens = new object[0] };

    // Act
    var respValido = await _fixture.HttpClient.PostAsJsonAsync("/api/pedidos", valido);
    var respInvalido = await _fixture.HttpClient.PostAsJsonAsync("/api/pedidos", invalido);

    // Assert
    respValido.StatusCode.Should().Be(HttpStatusCode.Created);
    respValido.Headers.Location.Should().NotBeNull();
    respInvalido.StatusCode.Should().Be(HttpStatusCode.BadRequest);
    (await respInvalido.Content.ReadAsStringAsync()).Should().Contain("vazio");
}
